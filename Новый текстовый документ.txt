<script>
    (function() {
        const submitBtn = document.getElementById('submitBtn');
        const resultPanel = document.getElementById('resultPanel');
        const scoreDisplay = document.getElementById('scoreDisplay');
        const gradeDisplay = document.getElementById('gradeDisplay');
        const feedbackDisplay = document.getElementById('feedbackDisplay');

        // Всего вопросов: 12 (каждый оценивается в 1 балл, кроме Q11 и Q12 — каждый правильный выбор = 0.25 балла, но суммируем до 1 балла)
        // Для простоты: Q1–Q10 — по 1 баллу; Q11 — 1 балл (все 4 верно), Q12 — 1 балл (все 5 верно)
        // Максимум: 12 баллов.

        const correctAnswers = {
            q1: 'b',
            q2: 'c',
            q3: 'b',
            q4: 'b',
            q5: 'c',
            q6: 'b',
            q7: 'b',
            q8: ['b', 'd', 'e'],
            q9: ['просмотр', 'просмотр.', 'чтение', 'только просмотр'],
            q10: 'c',
            // Q11: m1=a, m2=b, m3=c, m4=d
            m1: 'a',
            m2: 'b',
            m3: 'c',
            m4: 'd',
            // Q12: m5=a, m6=b, m7=c, m8=d, m9=e
            m5: 'a',
            m6: 'b',
            m7: 'c',
            m8: 'd',
            m9: 'e'
        };

        function getSelectedRadio(name) {
            const radios = document.getElementsByName(name);
            for (let r of radios) {
                if (r.checked) return r.value;
            }
            return null;
        }

        function getSelectedCheckboxes(name) {
            const checkboxes = document.getElementsByName(name);
            const selected = [];
            for (let cb of checkboxes) {
                if (cb.checked) selected.push(cb.value);
            }
            return selected;
        }

        function getSelectValue(name) {
            const sel = document.querySelector(`select[name="${name}"]`);
            return sel ? sel.value : null;
        }

        function evaluate() {
            let score = 0;
            let details = [];

            // 1
            const q1 = getSelectedRadio('q1');
            if (q1 === correctAnswers.q1) { score++; details.push('✅ Вопрос 1 — верно'); }
            else { details.push('❌ Вопрос 1 — неверно (правильно: электронное устройство для хранения, обработки, передачи и представления информации)'); }

            // 2
            const q2 = getSelectedRadio('q2');
            if (q2 === correctAnswers.q2) { score++; details.push('✅ Вопрос 2 — верно'); }
            else { details.push('❌ Вопрос 2 — неверно (правильно: процессор)'); }

            // 3
            const q3 = getSelectedRadio('q3');
            if (q3 === correctAnswers.q3) { score++; details.push('✅ Вопрос 3 — верно'); }
            else { details.push('❌ Вопрос 3 — неверно (правильно: операционная система)'); }

            // 4
            const q4 = getSelectedRadio('q4');
            if (q4 === correctAnswers.q4) { score++; details.push('✅ Вопрос 4 — верно'); }
            else { details.push('❌ Вопрос 4 — неверно (правильно: именованная область хранения данных)'); }

            // 5
            const q5 = getSelectedRadio('q5');
            if (q5 === correctAnswers.q5) { score++; details.push('✅ Вопрос 5 — верно'); }
            else { details.push('❌ Вопрос 5 — неверно (правильно: .docx)'); }

            // 6
            const q6 = getSelectedRadio('q6');
            if (q6 === correctAnswers.q6) { score++; details.push('✅ Вопрос 6 — верно'); }
            else { details.push('❌ Вопрос 6 — неверно (правильно: объединение одного или нескольких файлов в архив)'); }

            // 7
            const q7 = getSelectedRadio('q7');
            if (q7 === correctAnswers.q7) { score++; details.push('✅ Вопрос 7 — верно'); }
            else { details.push('❌ Вопрос 7 — неверно (правильно: архив — способ объединения и сжатия файлов, а резервная копия — дополнительная копия данных для восстановления)'); }

            // 8 (множественный)
            const q8selected = getSelectedCheckboxes('q8');
            const q8correct = correctAnswers.q8;
            const q8set = new Set(q8selected);
            const q8correctSet = new Set(q8correct);
            const q8isCorrect = q8set.size === q8correctSet.size && [...q8set].every(v => q8correctSet.has(v));
            if (q8isCorrect) { score++; details.push('✅ Вопрос 8 — верно'); }
            else { details.push('❌ Вопрос 8 — неверно. Правильные варианты: понятная структура папок и единые правила именования; удаление ненужных копий и контроль версий; разделение личных, учебных и рабочих материалов.'); }

            // 9 текстовый
            const q9input = document.querySelector('input[name="q9"]');
            const q9val = q9input ? q9input.value.trim().toLowerCase() : '';
            if (correctAnswers.q9.includes(q9val)) { score++; details.push('✅ Вопрос 9 — верно'); }
            else { details.push('❌ Вопрос 9 — неверно (правильно: просмотр / чтение)'); }

            // 10
            const q10 = getSelectedRadio('q10');
            if (q10 === correctAnswers.q10) { score++; details.push('✅ Вопрос 10 — верно'); }
            else { details.push('❌ Вопрос 10 — неверно (правильно: нельзя передавать свои пароли другим пользователям)'); }

            // 11 — соответствие компонентов
            let q11score = 0;
            const m1 = getSelectValue('m1');
            const m2 = getSelectValue('m2');
            const m3 = getSelectValue('m3');
            const m4 = getSelectValue('m4');
            if (m1 === correctAnswers.m1) q11score += 0.25;
            if (m2 === correctAnswers.m2) q11score += 0.25;
            if (m3 === correctAnswers.m3) q11score += 0.25;
            if (m4 === correctAnswers.m4) q11score += 0.25;
            if (q11score === 1) { score++; details.push('✅ Вопрос 11 — верно'); }
            else if (q11score > 0) { score += q11score; details.push(`⚠️ Вопрос 11 — частично верно (${q11score * 4}/4)`); }
            else { details.push('❌ Вопрос 11 — неверно'); }

            // 12 — соответствие форматов
            let q12score = 0;
            const m5 = getSelectValue('m5');
            const m6 = getSelectValue('m6');
            const m7 = getSelectValue('m7');
            const m8 = getSelectValue('m8');
            const m9 = getSelectValue('m9');
            if (m5 === correctAnswers.m5) q12score += 0.2;
            if (m6 === correctAnswers.m6) q12score += 0.2;
            if (m7 === correctAnswers.m7) q12score += 0.2;
            if (m8 === correctAnswers.m8) q12score += 0.2;
            if (m9 === correctAnswers.m9) q12score += 0.2;
            if (q12score === 1) { score++; details.push('✅ Вопрос 12 — верно'); }
            else if (q12score > 0) { score += q12score; details.push(`⚠️ Вопрос 12 — частично верно (${Math.round(q12score * 5)}/5)`); }
            else { details.push('❌ Вопрос 12 — неверно'); }

            // Округляем итоговый балл до 1 знака после запятой
            score = Math.round(score * 10) / 10;

            // Оценка (по 12-балльной шкале)
            let grade = '';
            if (score >= 11) grade = 'Отлично! 🌟';
            else if (score >= 8.5) grade = 'Хорошо! 👍';
            else if (score >= 6) grade = 'Удовлетворительно 📘';
            else grade = 'Неудовлетворительно 😕';

            // Вывод
            scoreDisplay.textContent = score;
            gradeDisplay.textContent = grade;

            let feedbackHtml = '<strong>📋 Разбор ответов:</strong><ul>';
            details.forEach(d => { feedbackHtml += `<li>${d}</li>`; });
            feedbackHtml += '</ul>';
            feedbackDisplay.innerHTML = feedbackHtml;

            resultPanel.classList.add('show');
            resultPanel.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
        }

        submitBtn.addEventListener('click', evaluate);

        const textInput = document.querySelector('input[name="q9"]');
        if (textInput) {
            textInput.addEventListener('keypress', (e) => {
                if (e.key === 'Enter') {
                    e.preventDefault();
                    evaluate();
                }
            });
        }
    })();
